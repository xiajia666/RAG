--------------------------------------------- 启动 Qdrant --------------------------------------------
# 创建数据目录
mkdir -p /data/qdrant

# 启动 Qdrant
docker run -d \
  --name qdrant \
  --restart=always \
  -p 6333:6333 \
  -p 6334:6334 \
  -v /data/qdrant:/qdrant/storage \
  qdrant/qdrant:latest
